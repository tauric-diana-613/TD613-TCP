const freeze=v=>{if(v&&typeof v==='object'&&!Object.isFrozen(v)){Object.values(v).forEach(freeze);Object.freeze(v);}return v;};
const finite=v=>Number.isFinite(Number(v));

function surface(name,episode_id,{value=null,measured=true}={}){return freeze({name,episode_id,value,measured});}
function byName(packet){return new Map((packet.surfaces||[]).map(s=>[s.name,s]));}
function sameEpisode(packet,names){const map=byName(packet);return names.every(name=>map.has(name)&&map.get(name).measured===true&&map.get(name).episode_id===packet.episode_id);}
function thresholdPass(packet){const map=byName(packet);const o=map.get('observer'),r=map.get('reconstruction'),j=map.get('joining');if(!o||!r||!j||![o.value,r.value,j.value].every(finite))return false;return Number(o.value)<=GOLDEN_EGG_THRESHOLDS.observer_leakage_bits&&Number(r.value)<=GOLDEN_EGG_THRESHOLDS.reconstruction_distance&&Number(j.value)<=GOLDEN_EGG_THRESHOLDS.joining_synergy_bits;}
